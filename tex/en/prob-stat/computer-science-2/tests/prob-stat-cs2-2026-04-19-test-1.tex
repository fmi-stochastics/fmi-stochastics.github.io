\begin{problem} (1.5 points)
\begin{enumerate}
\item (0.25 points) Let $\P(A)=0.5$, $\P(B\mid A)=0.4$ and $\P(A\cup B)=0.8$. Find $\P(B)$ and determine whether $A,B$ are independent.
\item (0.25 points) A discrete variable $X$ has probability generating function $g_X(s)=c(2+s)^4$. Find $c$, $\E[X]$ and $\P(X=2)$.
\item (0.25 points) Let $X,Y\overset{iid}{\sim}\mathrm{Geo}(p)$. Find $\P(X=Y)$.
\item (0.25 points) A fair coin is tossed 12 times. Let $X$ count heads in all 12 tosses and $Y$ heads in the first 5. Find $\mathrm{Cov}(X,Y)$ and $\rho_{X,Y}$.
\item (0.5 points) The joint distribution of $X,Y$ is
\[\begin{array}{c|ccc}Y\backslash X&-1&0&1\\\hline0&0.1&\alpha&0.2\\1&0.2&0.2&0.1\end{array}\]
Find $\alpha$, the marginal distributions, $\E[XY]$ and $\mathrm{Cov}(X,Y)$. Are $X,Y$ independent?
\end{enumerate}
\end{problem}
\begin{problem} (1.25 points)
After failing Probability, Angel turns to the underground life of a hacker and tries to breach the FMI server to change his grade. He sends malicious requests one after another. Each request is blocked by the firewall with probability $p\in(0,1)$, independently. The first unblocked request crashes the server and ends the attack. Let $N$ count the blocked requests before the breach.

Angel has forgotten that each blocked request leaves traces and generates warnings in the server logs. The number generated by request $i$ is $Y_i\sim\mathrm{Po}(\lambda)$, independent of the other requests and of $N$. Let $X=\sum_{i=1}^NY_i$ be the total warnings.
\begin{enumerate}
\item (0.25 points) Find $g_X(s)$.
\item (0.25 points) Find $\E[X]$ and $\mathrm{Var}(X)$.
\item (0.5 points) Find $\mathrm{Cov}(N,X)$.
\item (0.25 points) Given no warnings ($X=0$), what is the probability that Angel crashed the server with his first request ($N=0$)?
\end{enumerate}
\end{problem}
\begin{problem} (1.25 points)
Let $T\sim\mathrm{Ber}(0.4)$. Conditional on $T=1$, $X\sim\mathrm{Ge}(0.3)$; otherwise $X\sim\mathrm{Ge}(0.6)$. Here $\mathrm{Ge}(p)$ counts trials through the first success.
\begin{enumerate}
\item (0.25 points) Find $\P(X=k)$ for every $k\in\mathbb N$.
\item (0.25 points) Find $\P(T\leq0.4\mid X=1)$.
\item (0.25 points) Find $\E[X\mid X\geq2]$.
\item (0.5 points) Define $X_1,X_2$ similarly using $T_1,T_2$, where $T_1,T_2$ have the law of $T$ and are independent. Find $\P(X_1\leq X_2)$.
\end{enumerate}
\end{problem}
