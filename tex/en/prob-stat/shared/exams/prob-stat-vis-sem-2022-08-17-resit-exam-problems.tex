\begin{problem}
Let $(X_1,\dots,X_n)$ be a random permutation of the numbers in $\{1,2,\dots,n\}$, and let $S=X_1+\dots+X_n$.
\begin{enumerate}
\item (0.1 points) Find $\E S$ and $D S$.
\item (0.1 points) Prove that for two random variables $X$ and $Y$, $D(X+Y)=DX+DY+2Cov(X,Y)$.
\item (0.4 points) Express $\E S$ in terms of $\E X_i$. Find $\E X_i$, $\E X_i^2$ and $D X_i$ for every $i$.
\item (0.4 points) Express $D S$ in terms of $D X_i$ and $Cov(X_i,X_j)$. Find $Cov(X_i,X_j)$ for all $i,j$.
\end{enumerate}
\end{problem}
\begin{problem}
Let $U_1,V_1,U_2,V_2,\dots$ be independent $U(0,1)$ random variables, and let $M_i=\max(U_i,V_i)$.
\begin{itemize}
\item (0.75 points) Find $D(M_1+\dots+M_n)$ for every $n$.
\item (0.25 points) Give an approximation to the distribution of $M_1+\dots+M_n$ for large $n$.
\end{itemize}
\end{problem}
\begin{problem}
\begin{enumerate}
\item (0.5 points) Let $X_1,\dots,X_n$ be independent identically distributed random variables with density $f_X(x)=c/x^4$ for $x>1$, and 0 otherwise. If $S_n=X_1+\dots+X_n$, find constants $a,b,c$ such that $(S_n-a)/b$ is approximately standard normal, and justify your answer.
\item (0.5 points) Let $X_1,\dots,X_{40}$ be independent $Poi(10)$ random variables. Find a constant $a$ such that $X_1+\dots+X_{40}\in(a,\infty)$ with probability at least $99\%$.
\end{enumerate}
\end{problem}
